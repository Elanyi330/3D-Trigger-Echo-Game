# Weapons/Weapon_Resource.gd 新增（仅字段，不动既有逻辑）：
@export var noise_radius: float = 0.0   # 开火噪音半径（m）；0=静默

# Levels/M2_TDM/footstep_emitter.gd（新文件）
class_name FootstepEmitter
extends Node

signal footstep(pos: Vector3, noise_radius: float)

const RUN_RADIUS := 20.0     # m：跑动噪音半径（M3.6 参数化 .tres 迁移注记）

func setup(b: CharacterBody3D) -> void
func tick(delta: float) -> void   # 每物理帧（宿主 _physics_process 驱动）

# bot_perception.gd 扩展（T6 文件追加）：
const GUNSHOT_TTL := 3.0     # s：枪声事件衰减
const FOOTSTEP_TTL := 1.5    # s：脚步事件衰减
signal heard_event(kind: String, pos: Vector3)   # 敌对噪音被听到（距离 ≤ radius）
var _heard_events: Array = []   # [{pos, radius, t, kind}]
func _push_noise_event(kind: String, pos: Vector3, radius: float) -> void  # 事件注入（噪音总线接线 + 测试共用）

## 语义规格

1. **FootstepEmitter**：宿主 on_floor 且水平速度 > 0.5 时按 STRIDE 累计行进距离，每满
   2.5m 发 `footstep(pos, radius)`（速度 ≤ CROUCH_SPEED_MAX → CROUCH_RADIUS，否则
   RUN_RADIUS）。空中/静止不累计；落地/变速不清零累计距离（只在有前进时累计）。
2. **噪音事件注入**：`_push_noise_event(kind, pos, radius)` 追加 {pos, radius, t=_elapsed,
   kind}。枪声源接线：L_M2 中 WeaponManager 各 core 的 `shot_fired` 信号（既有，签名
   (ammo: int)）→ 对应 res.noise_radius > 0 时 `_push_noise_event("gunshot", 玩家位置,
   res.noise_radius)`——**M3.2 仅玩家枪声**（bot 开枪 M3.4 接入同源）；L_M2 装配一个
   噪音中继（建议：L_M2 直接连接各 core 信号到玩家 bot 的感知——敌 bot 的 perception
   在 M3.2 尚未装配（无 Brain），**本任务的接线点 = L_M2 预留**：实现一个
   `noise_bus`（Node + 信号 noise_event(kind, pos, radius)），L_M2 中玩家 core 的
   shot_fired → noise_bus；Grenade.exploded(center) → noise_bus("explosion", center, 50)。
   M3.3 装配敌 bot perception 时订阅 noise_bus（本任务测试直接调 _push_noise_event
   验证感知侧逻辑，bus 侧用单元测试验证接线函数）。
3. **tick 听觉节**（T6 视线节之后追加）：`_heard_events` 按 TTL 剔除（gunshot 3s /
   footstep 1.5s / explosion 3s）；新事件入队时若 `heard_at(body+EYE_OFFSET, pos,
   radius)` → `heard_event.emit(kind, pos)`。
4. **阵营过滤**：`_push_noise_event` 携带 source_faction 参数（默认 ""=无阵营不过滤——
   测试直调；噪音总线注入玩家事件时传 "friendly"）——敌 bot（faction="enemy"）听到
   friendly 源事件；同阵营源事件忽略（`_push_noise_event(kind, pos, radius, source)`。
   简化：事件条目存 source，tick 处理时 source==own_faction 跳过（M3.2 实际只有
   friendly 源 + enemy 监听组合）。
5. 时间戳口径：`_elapsed += delta` 于 tick 首行（T6 节流计时同源）。
