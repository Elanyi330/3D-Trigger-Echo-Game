# Levels/M2_TDM/bot_strategy.gd
class_name BotStrategy
extends Node

enum Strategy { ROAM, HUNTER, FLANKER, HOLD }   # 优先级：HUNTER > FLANKER > HOLD > ROAM

const DEATH_WINDOW := 15.0    # s：阵亡统计窗口（Hunter/Hold 触发）
const CLUSTER_RADIUS := 10.0  # m：阵亡簇判定（质心间最大距离）
const HUNTER_RADIUS := 25.0   # m：簇 C 半径内目标权重 ×3
const HUNTER_GUN_DRAW := 30.0 # m：进入 C 30m 切回主枪（T16 消费）
const GRENADE_RANGE := 20.0   # m：雷程（Hunter 突入前开路判定，T16 消费）
const HUNTER_LIFETIME := 45.0 # s
const HUNTER_QUIT_DIST := 20.0  # m：目标离开 C 区判定
const HUNTER_MAX := 2         # 同阵营 Hunter 占用上限（先到先得）
const FLANKER_MIN_DIST := 15.0   # m：目标直线距下限
const FLANKER_PATH_RATIO := 1.5  # 路径长/直线长阈值
const HOLD_TIMEOUT := 30.0    # s

const LOS_PATH_WEIGHT := 2.0  # Flanker LOS 段加权系数（路径成本语义，M3.3 实为
                              # 路线选择偏置：Brain 选点时对目标有 LOS 的方向加罚）

signal strategy_changed(from: int, to: int)

var faction: String            # 己方阵营（"enemy"/"friendly"——阵亡统计口径）
var board: EventBoard          # 死亡事件源 + 槽位（acquire_slot/release_slot）
var blackboard: BotBlackboard  # 读：state/target/target_lkp/alive_own/alive_enemy
var body: CharacterBody3D      # 位置/路径长来源（Flanker 评估）

func setup(f: String, b: EventBoard, bb: BotBlackboard, bd: CharacterBody3D) -> void
func tick(delta: float) -> void
func current() -> int           # Strategy 枚举（默认 ROAM）
func params() -> Dictionary     # 当前策略参数包（BotBrain 消费）

# static 纯函数（可单测，触发条件全量化）
static func death_cluster(deaths: Array, window: float, radius: float) -> Dictionary
# 输入 recent_deaths(own_faction, window) 的列表；输出
# {clustered: bool, centroid: Vector3, max_pair_dist: float}——
# ≥2 死亡且质心间最大距离 ≤ radius 即簇
static func hunt_target_bias(params: Dictionary, target_pos: Vector3,
        pos: Vector3) -> float   # Hunter：目标在 C radius 内 → 3.0，否则 1.0

        path_len: float) -> bool
static func hold_eligible(alive_own: int, alive_enemy: int, deaths_15s: int,
        clustered: bool) -> bool

# event_board.gd 槽位扩展（T15 授权）：
#   var strategy_slots: Dictionary = {}   # strategy_name -> count

## 语义规格

1. **触发**（tick 每物理帧评估；条件全部 static 纯函数）：
   - Hunter：`death_cluster(board.recent_deaths(faction, DEATH_WINDOW), 15, 10)
     .clustered` 且当前非 HUNTER 且冷却过且 `board.acquire_slot("hunter",
     HUNTER_MAX)` 成功 → HUNTER（槽位失败降级评估 FLANKER）
   - Flanker：`flanker_eligible(body.pos, 目标(LKP 优先), 路径长)`（路径长 =
     locomotion 当前路径剩余 + 已走——T15 内由黑板键 path_remaining 注入，
     Brain T14 写；缺键用直线距 ×2 保守估计，注释注明）
   - Hold：`hold_eligible(alive_own, alive_enemy, 15s 阵亡数, 非簇)`——簇 → Hunter
     优先（HUNTER 触发条件命中即不再评估 HOLD）
   - ROAM 恒真兜底
2. **裁决**：优先级 HUNTER > FLANKER > HOLD > ROAM；切换仅当新策略优先级高于当前
   （防横跳）；直接威胁优先——黑板 state==ENGAGE/CHASE 时冻结评估；同策略结束后
   COOLDOWN 20s 内不重复触发（每策略独立冷却计时）。
3. **生命周期**：
   - HUNTER：进入时黑板写 hunt_centroid；45s 超时 或 目标离开 C 区（目标位置距
     centroid > HUNTER_QUIT_DIST 连续 20s——连续计时在策略内）或 目标死亡 →
     ROAM + release_slot("hunter")
   - FLANKER：进入接敌（黑板 state 变 ENGAGE 或目标距 body < 30m）→ 结束回 ROAM
   - HOLD：30s 超时 或 alive_own ≥ alive_enemy → ROAM
4. **参数包**（params() 返回，BotBrain 消费；结构照抄）：
   ```
   ROAM:   {type: "roam"}
   HUNTER: {type: "hunter", centroid: Vector3, radius: 25.0, bias: 3.0,
            grenade_open: true}
   FLANKER:{type: "flanker", los_penalty: 2.0}
   HOLD:   {type: "hold", half_filter: float}   # half_filter = 己方半场 z 界
            #（faction=="enemy" → 南半场 z>0 保留；friendly → z<0——T14/Brain 消费）
   ```
5. **槽位**：EventBoard 共享单板（T9 装配）——Hunter 占用先到先得（同阵营多 bot
   共享板天然生效）；释放必须配对（离开 HUNTER 任一出口都 release）。
6. 策略内**无移动/武器调用**（不碰 locomotion/Brain 接口）；只读写黑板/事件板。

## 测试规格（test/unit/test_bot_strategy.gd，TDD RED 先行）

1. `test_death_cluster_positive`：3 死亡质心间距 ≤10 → clustered + centroid 正确
2. `test_death_cluster_negative`：间距 >10 或 <2 死亡 → 非簇
3. `test_hunt_bias_in_radius`：目标在 C 25m 内 → 3.0；外 → 1.0（边界 25.0）
4. `test_flanker_eligible_boundary`：直线 15/path 22.5（=1.5 倍）→ true；
   直线 14 或 path/直线 1.4 → false
5. `test_hold_eligible_conditions`：存活少 + ≥2 阵亡非簇 → true；簇 → false；
   存活持平 → false
6. `test_priority_arbitration`：当前 HUNTER 遇 FLANKER 候选 → 不切换；当前 ROAM
   遇 HOLD 候选 → 切换（注入黑板键/事件板构造触发条件）
7. `test_cooldown_blocks_retrigger`：HOLD 结束（超时）后 20s 内再造触发条件 → 不触发；
   21s 后 → 触发
8. `test_slot_cap_two`：3 个 strategy 实例（共享同一 EventBoard）依次申请 → 仅 2
   成功；一个 release 后第三个可 acquire
9. `test_hunter_lifetime`：HUNTER 推 45.1s → ROAM + 槽位释放（slots 计数回落）
10. `test_engage_freeze`：黑板 state 注入 ENGAGE → 触发条件满足也不切换（直接威胁
    优先铁律）
