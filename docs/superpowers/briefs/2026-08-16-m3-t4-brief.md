const WALL_PRESS_TIME := 0.5      # s：压墙判定（贴墙且水平速度 < 0.05 持续）
const WALL_SLIDE_TIME := 2.0      # s：切线滑移时长
const DEGEN_RUNWAY := 0.6         # m：退化跑道圆盘半径（目标点水平距 < 此值近静止放行）
const PROGRESS_TIMEOUT := 8.0     # s：无进展超时（位移 < 0.2 且未到达累计）
const REPATH_LIMIT := 2           # 次：重寻路上限（T2 已有同名概念——合并统一计数）
const STUCK_TIME := 5.0           # s：卡死判定窗口
const STUCK_DIST := 0.3           # m：卡死位移阈值（窗口内总位移 < 此值）
const KICK_NORMAL_TURN := 1.2     # rad：踢面正压转向角

# 实例状态（T4 追加）
var _wall_press_t := 0.0
var _wall_slide_t := 0.0
var _slide_handedness := 0        # 切线滑移手性锁存（每次 set_target 清零）
var _progress_t := 0.0
var _progress_pos := Vector3.ZERO
var _repath_count := 0            # 重寻路计数（T2/T3/T4 共享）
var _stuck_t := 0.0
var _stuck_pos := Vector3.ZERO

# static 纯函数（可单测）
static func slide_axis(body_yaw: float, wall_normal: Vector3, handedness: int) -> Vector2
# 返回与墙平行的本地移动轴（Vector2 x=左右/y=前后）：墙法向水平投影 n_h=(nx,0,nz)
# 切线 t = (n_z, 0, -n_x) × handedness；本地前向量 f = (sin yaw, 0, -cos yaw)；
# 输出 = 切线方向转为本地轴（x = t·本地右, y = t·本地前），归一化

        target_dir: Vector3) -> Vector2
# 踢面正压：接触法向水平投影 n_h，目标方向水平投影 d_h（均归一化）；
# 若 n_h·d_h <= -0.7（目标方向正对墙内）→ 返回沿 KICK_NORMAL_TURN 偏转后的方向
# 转本地轴（偏转方向 = 绕 +Y 转向 n_h 的正交侧，取与 d_h 夹角小的一侧）；否则返回
# d_h 转本地轴（正常前进）
static func progress_stalled(delta: float, moved: float, arrived_radius: float) -> bool
# 纯判定：moved < 0.2 且 > arrived_radius 时累加 delta，否则清零；返回累计 > PROGRESS_TIMEOUT

## 语义规格

1. **滑墙**（遍历器病理 1）：`body.get_slide_collision_count() > 0` 且水平速度 < 0.05 →
   `_wall_press_t` 累计；超 WALL_PRESS_TIME → `_wall_slide_t = WALL_SLIDE_TIME` 进入滑移，
   `_slide_handedness` 首次取 ±1 锁存（每次 set_target 清零）；滑移期 move_axis =
   `slide_axis(body.rotation.y, 碰撞法向, _slide_handedness)`；滑移结束 `_wall_press_t` 清零。
2. **踢面正压**（遍历器病理 2）：贴墙且 `kick_turn_axis` 判定目标方向正对墙内
   （n_h·d_h ≤ -0.7）→ move_axis 改用 `kick_turn_axis(...)` 输出（沿法向偏转 1.2rad
   脱离墙），替代直接前进；非正压时 kick_turn_axis 输出即正常前进（统一调用点）。
3. **可攀面豁免**：贴墙点前方 0 < 高差 ≤ 0.62（step-up 可走上）→ 不滑墙压入直走
   （探测：下一途经点 y − body y ∈ (0, 0.62] 即豁免）。
4. **退化跑道**：目标点水平距 < DEGEN_RUNWAY → 近静止放行（不触发超时/滑墙，
   到达判定正常收敛）。
5. **进展超时**：`_progress_t` 按 `progress_stalled` 语义累计（有位移清零）；超时 →
   `_repath_count < REPATH_LIMIT` 则 `set_target(原目标)` 重寻路 + 计数，超限 →
   `clear_target()`（诚实失败）。与 T3 跳跃失败的重寻路共用 `_repath_count`。
6. **卡死兜底**：`_stuck_t`/`_stuck_pos` 每 STUCK_TIME 窗口判定一次（窗口内总位移 <
   STUCK_DIST）→ 同上重寻路/放弃路径。
7. **正交性**：以上守卫仅在 WALK 段（T3 的 JUMP 段执行不套滑墙/踢面——跳跃有自己
   的 RUNUP 门控）；退化跑道/超时/卡死对 JUMP 段同样豁免（跳跃有 JUMP_TIMEOUT）。
